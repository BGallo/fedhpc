\section{The scalarised problem}
% =====================================================================

\begin{frame}{Two objectives, one scalar}
  \begin{itemize}
    \item FED-HPC schedules jobs onto resource \alert{types} (on-prem pools, cloud
      instance classes) at \alert{start times}, trading off two objectives:
    \begin{itemize}
      \item $f_1$ --- weighted turnaround: $\sum_j w_j\,(C_j - a_j)$, where
        $w_j$ is job $j$'s importance (given by the instance, default 1)
      \item $f_2$ --- monetary cost: $\sum_j c_{j,m}$ (on-demand cloud only)
    \end{itemize}
    \item The weighted-sum method collapses these into a single scalar,
      normalised by the reference points (\code{model.weighted\_sum\_terms}):
  \end{itemize}
  \begin{block}{Reference-point weighted sum}
    \[
      \min_x \;\; \lambda\,\frac{f_1(x) - f_1^{T}}{f_1^{0} - f_1^{T}}
                \;+\; (1-\lambda)\,\frac{f_2(x) - f_2^{0}}{f_2^{T} - f_2^{0}}
      \qquad \text{s.t.} \quad f_1(x) \le f_1^{0}
    \]
  \end{block}
  \begin{itemize}
    \item $f_1^{T}$: lex-min turnaround \quad $f_2^{T}$: min cost at that
          turnaround \quad $(f_1^{0}, f_2^{0})$: lex-min $(f_2, f_1)$ point
          ($f_2^{0}{=}0$ if a free schedule exists)
    \item $\lambda \in [0,1]$ is the trade-off knob; $\lambda{=}1$ is pure turnaround
  \end{itemize}
\end{frame}

\begin{frame}{The metaheuristic's fitness}
  \begin{itemize}
    \item The wrapper converts $\lambda$ into internal weights
      $w_1 = \lambda/(f_1^{0}{-}f_1^{T})$, $w_2 = (1{-}\lambda)/(f_2^{T}{-}f_2^{0})$,
      soft cap $f_1^{\text{cap}} = f_1^{0}$ (constant offset dropped)
    \item The solver then minimises a \alert{penalised scalar fitness} \g:
  \end{itemize}
  \begin{block}{$G(x)$ --- \code{struct WeightedScalar}}
    \[
      G(x) = w_1 f_1(x) + w_2 f_2(x)
             + p_{\text{cap}} \big(f_1(x) - f_1^{\text{cap}}\big)^{+}
             + \big[10^{9} + 10^{6}\,cv(x)\big]\cdot \mathbb{1}\{cv(x) > 0\}
    \]
  \end{block}
  \begin{itemize}
    \item $p_{\text{cap}} = 10^{3}\,w_1$ ($10^{3}\,w_2$ when $w_1{=}0$) --- soft
      Pareto-region penalty
    \item $cv(x)$ --- aggregate on-prem capacity violation (no budget constraint)
    \item Feasibility is lexicographically dominant, but infeasible individuals
      stay \emph{ordered} by violation --- selection always has a gradient
  \end{itemize}
\end{frame}

\section{The decoder}
% =====================================================================

\begin{frame}{\code{schedule\_repair} --- SPT list scheduling}
  \begin{itemize}
    \item Jobs are sorted \alert{shortest-processing-time first} (ties by
      current start) and each is moved to the \textbf{earliest capacity-feasible}
      slot on its current type
    \item \code{job\_slots[j]} is start-sorted within a type, so the first
      feasible slot found is the earliest --- $f_1$ is non-increasing,
      capacity violations only shrink, cost is invariant
    \item \textbf{Free-pool balancing} (default on): widen the scan to every
      type that \emph{weakly dominates} the current one (cost~$\le$ and
      $p_{\text{occ}}\le$) --- spreads load across equally priced pools instead
      of queueing everyone on the one the gene names
  \end{itemize}
  \begin{alertblock}{Why SPT alone is not enough}
    SPT is optimal for $P\,\|\,\sum C_j$ (parallel machines, \emph{no} release
    dates) but carries \textbf{no guarantee} once jobs arrive over time --- and
    under heavy capacity contention that gap becomes real (next slides).
  \end{alertblock}
\end{frame}

\begin{frame}{\code{local\_search} --- congestion repair \& polish}
  \begin{itemize}
    \item Bounded coordinate descent, restricted to a per-job, per-type
      candidate shortlist (\code{job\_candidates}: nearest few slots per type,
      plus geometrically spaced positions across the horizon)
    \item For each job:
    \begin{itemize}
      \item a candidate that \textbf{strictly reduces} the job's marginal
        capacity violation is taken unconditionally --- repair always wins
      \item among candidates tied on violation, only a \textbf{weakly
        Pareto-dominating} one is taken --- never regresses either objective
    \end{itemize}
    \item Together with \code{schedule\_repair} this realises the
      decomposition: the repair relocates whole congested populations of jobs
      across the horizon in one sweep; local search mops up residual
      congestion and descends flat iso-$f_1$ plateaux
  \end{itemize}
\end{frame}

\begin{frame}{Extract-from-Preempt re-decode \hfill{\small\code{decode\_order=1}}}
  \begin{itemize}
    \item Fix the type vector: the start-time subproblem is
      $P \mid r_j \mid \sum C_j$ --- strongly NP-hard; SPT has no guarantee
      once jobs carry release dates
    \item \textbf{\code{srpt\_completion\_order}}: simulate the optimal
      \emph{preemptive} per-type schedule (SRPT --- always run the cap-many
      released jobs of least remaining work); record completion times
    \item \textbf{Re-decode}: reset genes to their latest feasible slot, then
      list-schedule in ascending preemptive-completion order --- the
      \alert{Extract-from-Preempt} 2-approximation (Phillips, Stein \& Wein, 1998)
    \item Keep the lower-$f_1$ of \{SPT, SRPT\} --- $f_2$ untouched, \g monotone
  \end{itemize}
  \begin{block}{Now the default}
    Promoted by an A/B: neutral without congestion, 1--5~pp lower gap to the
    proven front under heavy contention, seed-stable, $\sim\!1.5\times$ time.
  \end{block}
\end{frame}

\section{Embedded local search}
% =====================================================================

\begin{frame}{\code{weighted\_local\_search} --- the memetic core}
  \begin{itemize}
    \item A \alert{steepest-descent hill climb} over single job$\to$type
      reassignments, each decoded exactly --- the objective is scalar, so
      there is one improving direction and the search genuinely reaches a
      local optimum of \g
    \item Candidate moves are ranked by an $O(1)$ estimate before the
      expensive exact decode:
  \end{itemize}
  \[
    \widehat{\Delta g} = w_1\big(p'_{\text{occ}} - p_{\text{occ}}\big)
                        + w_2\big(\text{cost}' - \text{cost}\big)
  \]
  \begin{itemize}
    \item $\Delta f_2$ exact; $\Delta f_1$ approximated by $\Delta p_{\text{occ}}$
      at the target type's earliest slot
    \item Only the \code{shortlist} most promising moves are decoded exactly;
      the true best is committed --- \textbf{Lamarckian}: improvements are
      written back into the genotype
  \end{itemize}
\end{frame}

\begin{frame}{\code{weighted\_local\_search} --- procedure}
  \footnotesize
  \begin{algorithmic}[1]
    \Procedure{WeightedLocalSearch}{$x$, \texttt{maxMoves}, \texttt{shortlist}}
      \State $x \leftarrow$ \Call{ScheduleRepair}{$x$}; \Call{Evaluate}{$x$}; $\;g \leftarrow G(x)$
      \For{$mv = 1 \ldots \texttt{maxMoves}$}
        \State candidates $\leftarrow$ every job$\to$type flip with $\widehat{\Delta g} < 0$
        \State keep the \texttt{shortlist} candidates of least $\widehat{\Delta g}$
        \ForAll{kept candidate $c$}
          \State decode $c$ exactly: \Call{ScheduleRepair}{}, \Call{LocalSearch}{}, \Call{Evaluate}{}
        \EndFor
        \If{best decoded $G < g$}
          \State $x \leftarrow$ best decoded; $g \leftarrow G(x)$
        \Else\ \textbf{break}
        \EndIf
      \EndFor
      \State \Return $x$
    \EndProcedure
  \end{algorithmic}
\end{frame}

\section{Population \& evolution}
% =====================================================================

\begin{frame}{Initialisation --- structured seeds}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.5\textwidth}
      \small
      \textbf{Structured} (\code{make\_seeds}):
      \begin{itemize}
        \item greedy min-turnaround / min-cost
        \item as-soon-as-possible
        \item cloud-burst
        \item fixed-delay 25\% / 50\%
        \item index-staggered
        \item capacity-aware first-fit list schedule
      \end{itemize}
    \end{column}
    \begin{column}{0.5\textwidth}
      \small
      \textbf{Randomised remainder} (\code{make\_random}):
      \begin{itemize}
        \item each gene drawn from a cost-biased distribution over the
          job's slots
        \item cheaper slots weighted higher (worst slot keeps
          20\% of the best slot's weight)
        \item degenerates to uniform when there is no cost signal
      \end{itemize}
    \end{column}
  \end{columns}
  \vspace{1.5mm}
  \begin{alertblock}{Every seed is refined}
    Each individual is run through \code{weighted\_local\_search} before
    evolution begins --- the initial population is a population of local optima.
  \end{alertblock}
\end{frame}

\begin{frame}{The generational loop --- skeleton}
  \footnotesize
  \begin{algorithmic}[1]
    \State $\text{pop} \gets$ heuristic seeds $\cup$ weighted-random; refine each with \Call{WeightedLocalSearch}{}
    \State $\text{incumbent} \gets \arg\min_x G(x)$;\quad $\text{stagn} \gets 0$
    \For{$\text{gen} = 1 \ldots$ \texttt{n\_gen}}
      \State $\text{offspring}[0] \gets \text{incumbent}$ \Comment{elitism}
      \State $a,b \gets$ \Call{Tournament}{pop}, \Call{Tournament}{pop} \Comment{$\forall k$, parallel}
      \State $c \gets$ \Call{MSXF}{$a,b,G$};\ \Call{Mutate}{$c$};\ \Call{WeightedLocalSearch}{$c$}
      \State $\text{offspring}[k] \gets c$;\quad $\text{pop} \gets \text{offspring}$
      \If{improved}
        \State update incumbent; $\text{stagn}\gets 0$
      \Else
        \State $\text{stagn}\gets\text{stagn}+1$
      \EndIf
      \If{$\text{stagn} \ge$ \texttt{restart\_patience}}
        \State \Call{ILSKick}{}
      \EndIf
    \EndFor
    \State \Return \Call{WeightedLocalSearch}{incumbent, $3\times$ budget} \Comment{final polish}
  \end{algorithmic}
\end{frame}

\begin{frame}{Recombination --- Multi-Step Crossover Fusion}
  \begin{itemize}
    \item \alert{MSXF} (Yamada \& Nakano) is a short, guided walk through
      type-assignment space \emph{from} parent $b$ \emph{toward} parent $a$,
      not an index-level crossover
    \item Each of up to \code{steps} iterations:
    \begin{enumerate}
      \item sample (up to 12) genes where the parents still differ
      \item adopt --- after an SPT decode --- the single gene of $a$ that
        most reduces $G$
      \item record the best point met along the path
    \end{enumerate}
    \item A short local search \emph{between} the parents: never shreds a
      coherent building block the way index crossover can
    \item \code{xover\_mode}: 0 = two-point, 1 = none (pure multi-start ILS),
      \textbf{2 = MSXF (default)}
  \end{itemize}
  \begin{alertblock}{A/B on the two 10-min instances}
    gap to the Gurobi optimum \; $+2.0\%\!\to\!+0.3\%$ \; / \; $+3.9\%\!\to\!+3.5\%$
    \; vs.\ plain two-point crossover.
  \end{alertblock}
\end{frame}

\begin{frame}{Mutation}
  \begin{itemize}
    \item \textbf{Default (\code{mut\_mode=0}):} uniform resample --- each gene,
      independently with probability $p_{\text{mut}} = 2/n_{\text{jobs}}$, drawn
      uniformly from the job's slots
    \item \textbf{Scalar-directed reinsert (\code{mut\_mode=1},} Feltl \& Raidl
      style\textbf{):} reinsert on a strictly \emph{cheaper} type with
      probability $w_2/(w_1{+}w_2)$, otherwise a strictly \emph{faster} type ---
      biasing perturbation toward whichever objective the current
      $\lambda$-subproblem weights more heavily
    \item Roughly neutral once MSXF + local search carry most of the search;
      default stays at uniform resample
  \end{itemize}
\end{frame}

% =====================================================================
\section{Diversification}
% =====================================================================

\begin{frame}{ILS perturbation kicks}
  \begin{itemize}
    \item A stagnation counter tracks consecutive generations without an
      improving incumbent
    \item On reaching \code{restart\_patience} (default \textbf{6}):
    \begin{enumerate}
      \item perturb the incumbent with $\max(3, n_{\text{jobs}}/40)$ random
        single-job type reassignments
      \item drive it back to local optimality with a full-budget
        \code{weighted\_local\_search}
      \item inject it in place of the population's weakest member; adopt as
        incumbent if it improves $G$
      \item reset the stagnation counter
    \end{enumerate}
    \item The canonical \alert{perturb $\to$ local-search $\to$ accept} cycle
      of iterated local search, applied to the elite solution while the
      population keeps supplying recombinative diversity
    \item \code{restart\_patience = 0} disables the mechanism
  \end{itemize}
\end{frame}

\begin{frame}{What kind of algorithm is this?}
  \begin{center}
  \small
  \begin{tabular}{@{}lll@{}}
    \toprule
    \textbf{layer} & \textbf{role} & \textbf{technique} \\
    \midrule
    Encoding      & search-space reduction         & type-vector genotype + SPT/SRPT decoder \\
    Local search  & intensification                & steepest-descent type-flip hill climb \\
    Population    & recombinative diversification  & elitist GA, tournament, MSXF \\
    ILS kicks     & escaping local optima          & perturb + local search + elitist accept \\
    Constraints   & feasibility                    & static lexicographic exterior penalty \\
    \bottomrule
  \end{tabular}
  \end{center}
  \vspace{2mm}
  \begin{alertblock}{A memetic algorithm with an indirect encoding}
    The expensive combinatorial work (start-time scheduling under capacity)
    is delegated to a polynomial decoder; evolution and local search operate
    on the much smaller type-assignment space, and recombination is itself a
    guided local search between elite solutions.
  \end{alertblock}
\end{frame}

\section{The dispatch memetic}
% =====================================================================

\begin{frame}{Same GA, different representation \hfill{\small\code{weighted\_memetic.hpp}}}
  \begin{itemize}
    \item The slot genome wastes its search on \alert{pool$\leftrightarrow$pool
      plateaus}: on the 10-min instances every on-prem type is equivalent per
      job (same $p_{\text{occ}}$, cost~0)
    \item \code{weighted\_solve\_memetic} keeps the control flow of
      \code{run\_weighted} (seeds, tournament, MSXF, mutation, capped LS,
      elitism, ILS kicks) and swaps the three problem-specific parts:
  \end{itemize}
  \begin{center}
  \footnotesize
  \begin{tabular}{@{}p{1.6cm}p{4.6cm}p{6.6cm}@{}}
    \toprule
    & \textbf{slot GA} (\code{weighted.hpp}) & \textbf{dispatch memetic} \\
    \midrule
    genome  & slot index per job & $0$ = auto, $1$ = cloud, $2$ = on-prem \\
    decoder & SPT repair + SRPT re-decode & time-driven non-delay dispatch: waiting
              jobs in priority order start on any free pool; \emph{auto} jobs
              are offloaded once waiting longer scores worse than cloud
              (deadline rule) \\
    LS      & type-flip hill climb & \code{dispatch\_ls}: single on-prem/cloud
              flips, fast \code{DEngine} decode \\
    \bottomrule
  \end{tabular}
  \end{center}
\end{frame}

\begin{frame}{Dispatch priorities \& self-adaptive settings}
  \begin{itemize}
    \item Priority $v_j = (w_1 w_j + \mu\,\text{regret}_j)/p_j$, highest first
      ($\mu{=}0$ is WSPT; larger $\mu$ favours jobs whose cloud option is dear;
      $\text{regret}_j$ is defined in ``EDF, step 2'')
    \item \alert{EDF} (\code{flags} 4096): an alternative priority, earliest
      \emph{indifference deadline} first --- explained in the next three slides
    \item \code{adapt} bits make decoder settings \emph{per individual}:
      spread over the initial population, inherited by crossover (MSXF treats
      ``adopt the other parent's settings'' as one more step), mutated ---
      1 = $\mu$-grid index, 2 = deadline offload on/off, 4 = dispatch /
      serial-SGS decoder, 8 = LS also tries neighbouring settings
    \item LS decodes with the fast \code{DEngine}; scores are re-summed in job
      order (\code{canon}) so fast and reference decoders follow
      bit-identical trajectories
  \end{itemize}
\end{frame}

\begin{frame}{EDF, step 1 --- what one job adds to the objective}
  \begin{itemize}
    \item The solver minimises one number,
      $g = w_1 f_1 + w_2 f_2$ (plus penalties; slide ``The metaheuristic's
      fitness''). Two kinds of weight appear --- do not confuse them:
  \end{itemize}
  \begin{center}
  \footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    symbol & meaning & set by \\
    \midrule
    $w_1$ & price of one unit of weighted turnaround in $g$
          ($=\lambda/(f_1^0{-}f_1^T)$) & $\lambda$ + reference points \\
    $w_2$ & price of one unit of money in $g$
          ($=(1{-}\lambda)/(f_2^T{-}f_2^0)$) & $\lambda$ + reference points \\
    $w_j$ & importance of job $j$ inside $f_1$ & the instance \\
    $a_j$, $r_j$ & arrival of $j$; earliest time it can start on-prem & the instance \\
    $p_j$, $c_j$ & runtime and money cost of $j$ on a given machine type & the instance \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{itemize}
    \item $g$ is a sum over jobs, so each job has its \alert{own score}:
      $\;g_j = w_1\, w_j\,(C_j - a_j) + w_2\, c_j$
    \item \textbf{On-prem}, started at time $s$: \quad
      $g^{\text{on}}_j(s) = w_1 w_j\,(s + p_j - a_j) + w_2\, c^{\text{on}}_j$
      \quad --- every unit of waiting adds $w_1 w_j$
    \item \textbf{Cloud}: starts at once (cloud capacity is unlimited), so its
      score $g^{\text{cl}}_j$ is a \emph{constant}, known before scheduling
  \end{itemize}
\end{frame}

\begin{frame}{EDF, step 2 --- the indifference deadline}
  \begin{columns}[T,onlytextwidth]
    \begin{column}{0.54\textwidth}
      \begin{itemize}
        \item $\text{regret}_j = g^{\text{cl}}_j - g^{\text{on}}_j(r_j)$: how
          much \emph{worse} cloud is than on-prem started as early as possible
        \item Each unit of waiting costs $w_1 w_j$. Once the regret is used
          up, both options score the same: that start time is the
          \alert{indifference deadline} $d_j$. Later, cloud is better and the
          decoder offloads the job
        \item \textbf{EDF} (\emph{Earliest Deadline First}): when a pool has
          room, start the waiting job with the smallest $d_j$ --- the one
          closest to being better off in the cloud
      \end{itemize}
    \end{column}
    \begin{column}{0.44\textwidth}
      \centering
      \begin{tikzpicture}[x=0.62cm, y=0.55cm, font=\scriptsize]
        \draw[-{Stealth[length=1.8mm]}] (0,0) -- (6.3,0) node[below left]{start $s$};
        \draw[-{Stealth[length=1.8mm]}] (0,0) -- (0,4.6) node[left]{$g_j$};
        \draw[mDarkTeal, very thick] (0.8,1.0) -- (5.6,4.2)
          node[above, align=center]{on-prem $g^{\text{on}}_j(s)$\\[-1mm]slope $w_1 w_j$};
        \draw[mLightBrown, very thick] (0.8,2.6) -- (6.0,2.6);
        \node[mLightBrown, above right, inner sep=1pt] at (0.8,2.6) {cloud $g^{\text{cl}}_j$};
        \draw[dashed] (3.2,2.6) -- (3.2,0) node[below]{$d_j$};
        \draw[dashed] (0.8,1.0) -- (0.8,0) node[below]{$r_j$};
        \draw[{Stealth[length=1.5mm]}-{Stealth[length=1.5mm]}] (0.55,1.0) -- (0.55,2.6)
          node[midway, left]{regret$_j$};
        \fill (3.2,2.6) circle (1.3pt);
        \node[mDarkTeal, align=center] at (2.0,4.1) {on-prem\\[-1mm]better};
        \node[mLightBrown, align=center] at (4.8,1.0) {cloud\\[-1mm]better};
      \end{tikzpicture}
      \[
        d_j = \min\Big(\ell_j,\; r_j + \frac{\max(\text{regret}_j, 0)}{w_1 w_j}\Big)
      \]
      \scriptsize $\ell_j$: latest on-prem start; it is the deadline
      when $j$ has no cloud option or $w_1{=}0$.
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{EDF, step 3 --- a worked example}
  \textbf{One job $j$}: $w_1 = 1$, weight $w_j = 2$, arrives at $a_j = r_j = 0$,
  runs $p_j = 3$ on both sides; on-prem is free, the cloud run's money
  adds $w_2 c = 8$ to $g$.
  \begin{center}
  \footnotesize
  \begin{tabular}{@{}lll@{}}
    \toprule
    & formula & value \\
    \midrule
    cloud score         & $w_1 w_j\,p_j + w_2 c = 1\cdot2\cdot3 + 8$ & $g^{\text{cl}}_j = 14$ \\
    on-prem, start $s$  & $w_1 w_j\,(s + p_j - a_j) = 2(s + 3)$        & $g^{\text{on}}_j(0) = 6$ \\
    regret              & $14 - 6$                                     & $8$ \\
    deadline            & $r_j + \text{regret}/(w_1 w_j) = 0 + 8/2$    & $d_j = \mathbf{4}$ \\
    \bottomrule
  \end{tabular}
  \end{center}
  \begin{itemize}
    \item On-prem start $s=2$: $2\cdot5 = 10 < 14$, on-prem better;
      \; $s=4$: $2\cdot7 = 14$, tie ($=d_j$);
      \; $s=6$: $2\cdot9 = 18 > 14$, cloud better
    \item So if $j$ is still waiting at $t = 4$, the decoder offloads it
    \item A heavier job ($w_j$ larger) or a cheaper cloud (regret smaller)
      gets an \alert{earlier} deadline, so EDF starts it sooner
  \end{itemize}
\end{frame}

\begin{frame}{Why EDF: WSPT starves long jobs into the cloud}
  \begin{itemize}
    \item The decoder offloads a waiting \emph{auto} job once it passes $d_j$.
      Under \textbf{WSPT/$\mu$} order on congested pos instances, long
      low-priority jobs wait past their deadlines and are offloaded one by one
    \item Each offload is locally rational; together they cost
      \alert{20$\times$} the optimum's cost term (0.277 vs 0.014,
      A\_pos\_n300\_lH, $\lambda{=}0.25$). A gene flip to on-prem keeps the
      order, so the job just waits longer
    \item EDF serves each job \emph{before} cloud becomes cheaper; $d_j$
      tracks the optimum's on-prem start (Spearman $\rho = 0.6$--$0.9$).
      Median gap at $\lambda{=}0.25$: \textbf{30\,\% $\to$ 9\,\%}
    \item \textbf{How the solvers use it:} as an \emph{extra} priority, since
      EDF loses at $\lambda{=}0.75$ where the optimum offloads most jobs; in
      the GA it is one more value of the evolved priority index ($\mu=-2$).
      With \code{flags} 8192, EDF individuals skip the deadline offload (late
      jobs wait on-prem); keeping the offload raises GA-mid's gap 4.5 $\to$
      5.5\,\%
  \end{itemize}
\end{frame}

\begin{frame}{Why EDF needs converged seeds}
  \begin{alertblock}{The first EDF port still lost to the ILS (2026-11-04)}
    The LS is capped at 4--8 moves per individual. From all-auto, the EDF
    start needs \textbf{166--1634} moves; the $\mu$ start needs 3--46. After
    the initial LS, EDF individuals sit at 60--128\,\% gap and selection
    removes them by generation 0--1.
  \end{alertblock}
  \begin{itemize}
    \item The ILS does not suffer: each of its starts gets a full LS and its
      own kicks before being compared
    \item Fixes, as \code{adapt} bits:
    \begin{itemize}
      \item \textbf{16} --- all-auto seed individuals, LS run to a local
        optimum (K~64 / K\textsubscript{max}~256)
      \item \textbf{128} --- with 16, seed only the ILS's two starts (best
        $\mu$ by all-auto decode, EDF)
      \item \textbf{64} --- niche elitism: each priority's best survives
        every generation
      \item 32 --- final polish to a local optimum (not used)
    \end{itemize}
    \item The GA then starts from the ILS's converged starts and
      recombination improves on them
  \end{itemize}
\end{frame}

\section{Summary}
% =====================================================================

\begin{frame}{Summary}
  \begin{itemize}
    \item An \alert{indirect-encoding memetic algorithm}: search the
      type-assignment space, \textbf{decode} start times with a
      capacity-aware list scheduler
    \item \textbf{Two decoders}, lower $f_1$ kept: SPT and
      \textbf{Extract-from-Preempt} (release-date aware, default)
    \item \textbf{Local search} intensifies (Lamarckian type-flip hill
      climbing); \textbf{MSXF} is a guided search between parents;
      \textbf{ILS kicks} escape stagnation
    \item The \textbf{dispatch memetic} runs the same GA on an
      auto/cloud/on-prem genome and a dispatch decoder; with EDF and
      converged seeds it beats the dispatch ILS from $\sim$3\,s per cell
    \item Deterministic, parallel, validated against the exact Gurobi MIP
      and by ablation
  \end{itemize}
